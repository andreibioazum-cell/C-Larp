# Модули игры в порядке компиляции. gen.py читает этот список, пути указаны
# относительно game/scripts. Порядок важен для глобальных объявлений: классы и
# их экземпляры должны стоять раньше первого использования, поэтому сначала
# константы и состояние, потом мир, раунд и лобби, а замыкает список движок -
# главный цикл. Файл .ds, которого нет в списке, всё равно компилируется, но
# после всех перечисленных и по алфавиту. Текст после # - комментарий.

# core: константы, состояние и общие помощники
core/config.ds                    # константы: фазы, тайминги, баланс, палитра, строки
core/state.ds                     # класс Fighter и глобальные поля фазы
core/ui.ds                        # помощники интерфейса: текст, плашки, таймер
core/textures.ds                  # загрузка спрайтов и музыка лобби

# world: арена и боец
world/arena.ds                    # пол, стены, столбы, столкновения
world/fighter.ds                  # движение, откаты, отрисовка бойца

# round: правила раунда
round/attack.ds                   # удар убийцы и попадания
round/bot.ds                      # бот: погоня, побег, прогулка
round/round.ds                    # лобби -> раунд -> итог -> лобби
round/render.ds                   # сцена, HUD и плашки раунда

# lobby: перерыв
lobby/lobby.ds                    # обратный отсчёт и итог прошлого раунда

# ввод и движок
core/input.ds                     # джойстик, кнопка удара, системная "назад"
core/engine.ds                    # init, update, draw
